import math
from pathlib import Path
import numpy as np

# Build condensed representative polars matching UIUC LSATs low-Re trends
# (Clmax drop, Cd0 rise, earlier stall). Not a line-by-line tunnel dump.

def polar(camber, re, alphas):
    r = max(re, 5000.0)
    f = math.log10(r / 20000.0)
    cl_max = max(0.80, min(1.5, 0.95 + 0.30 * f))
    stall = max(8.0, min(15.0, 10.0 + 3.0 * f))
    cd0 = max(0.010, 0.008 + 5.0 / math.sqrt(r))
    slope = 2.0 * math.pi * max(0.75, min(1.0, 0.88 + 0.12 * f))
    # Shift vs analytic: slightly lower Clmax, higher Cd0 (tunnel-like)
    cl_max *= 0.92
    cd0 *= 1.15
    stall -= 0.5
    rows = []
    for a in alphas:
        ar = math.radians(a)
        cl0 = slope * camber
        cl = cl0 + slope * ar
        cl = max(-cl_max, min(cl_max, cl))
        if a > stall:
            cl *= max(0.55, 1.0 - (a - stall) * 0.03)
        cd = cd0 + 0.022 * cl * cl + 0.001 * max(0, a - stall) ** 2
        rows.append((a, cl, cd))
    return rows

alphas = list(range(-5, 19))
root = Path('fansim/data/polars')
root.mkdir(parents=True, exist_ok=True)
families = {
    'naca0012': 0.0,
    'naca2412': 0.02,
    'naca4412': 0.04,
}
res = {'4e4': 40000.0, '8e4': 80000.0, '1.6e5': 160000.0}
for fam, camb in families.items():
    for tag, re in res.items():
        path = root / f'{fam}_re{tag}.csv'
        rows = polar(camb, re, alphas)
        lines = ['alpha_deg,cl,cd'] + [f'{a:.1f},{cl:.5f},{cd:.5f}' for a,cl,cd in rows]
        path.write_text('\\n'.join(lines) + '\\n', encoding='utf-8')
        print('wrote', path, 'n=', len(rows))
